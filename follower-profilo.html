/** * ============================================================================ * SERVER UNICO INTEGRATO: Dashboard Creator/Follower + Stripe Connect V2 * ============================================================================ */ const express = require('express'); const Stripe = require('stripe'); const app = express(); app.use(express.json()); app.use(express.urlencoded({ extended: true })); // ========================================== // CONFIGURAZIONE STRIPE // ========================================== const STRIPE_SECRET_KEY = process.env.STRIPE_SECRET_KEY || 'sk_test_YOUR_KEY_HERE'; const WEBHOOK_SECRET = process.env.STRIPE_WEBHOOK_SECRET || 'whsec_YOUR_WEBHOOK_SECRET'; if (!STRIPE_SECRET_KEY || STRIPE_SECRET_KEY === 'sk_test_YOUR_KEY_HERE') { console.warn('⚠️ Attenzione: Chiave Stripe non configurata. Imposta la variabile d\'ambiente STRIPE_SECRET_KEY.'); } const stripeClient = new Stripe(STRIPE_SECRET_KEY); // Database in-memoria per la gestione locale const db = { connectedAccounts: new Map(), // userId -> Stripe Account ID products: [] // Prodotti salvati }; // ========================================== // API BACKEND STRIPE // ========================================== app.post('/api/create-account', async (req, res) => { try { const { userId, displayName, contactEmail } = req.body; const account = await stripeClient.v2.core.accounts.create({ display_name: displayName || 'Creator Profile', contact_email: contactEmail || 'creator@example.com', identity: { country: 'us' }, dashboard: 'express', defaults: { responsibilities: { fees_collector: 'application', losses_collector: 'application' } }, configuration: { recipient: { capabilities: { stripe_balance: { stripe_transfers: { requested: true } } } } } }); db.connectedAccounts.set(userId, account.id); res.json({ success: true, accountId: account.id }); } catch (error) { console.error('Errore account:', error); res.status(500).json({ error: error.message }); } }); app.post('/api/create-account-link', async (req, res) => { try { const { accountId } = req.body; const accountLink = await stripeClient.v2.core.accountLinks.create({ account: accountId, use_case: { type: 'account_onboarding', account_onboarding: { configurations: ['recipient'], refresh_url: `http://localhost:4280/?accountId=${accountId}`, return_url: `http://localhost:4280/?accountId=${accountId}&onboarded=true`, }, }, }); res.json({ url: accountLink.url }); } catch (error) { console.error('Errore link:', error); res.status(500).json({ error: error.message }); } }); app.post('/api/create-product', async (req, res) => { try { const { name, description, unitAmount, currency, connectedAccountId } = req.body; const product = await stripeClient.products.create({ name, description, default_price_data: { unit_amount: parseInt(unitAmount, 10), currency: currency || 'usd' }, }); const productRecord = { id: product.id, name: product.name, description: product.description, priceId: product.default_price, unitAmount, currency: currency || 'usd', connectedAccountId, }; db.products.push(productRecord); res.json({ success: true, product: productRecord }); } catch (error) { console.error('Errore prodotto:', error); res.status(500).json({ error: error.message }); } }); app.get('/api/storefront-data', (req, res) => { const accountsList = Array.from(db.connectedAccounts.entries()).map(([userId, accountId]) => ({ userId, accountId })); res.json({ products: db.products, connectedAccounts: accountsList }); }); app.post('/api/create-checkout-session', async (req, res) => { try { const { priceId, connectedAccountId } = req.body; const session = await stripeClient.checkout.sessions.create({ line_items: [{ price: priceId, quantity: 1 }], payment_intent_data: { application_fee_amount: 150, // Trattenuta piattaforma ($1.50) transfer_data: { destination: connectedAccountId }, }, mode: 'payment', success_url: `http://localhost:4280/?success=true`, cancel_url: `http://localhost:4280/?canceled=true`, }); res.json({ url: session.url }); } catch (error) { console.error('Errore checkout:', error); res.status(500).json({ error: error.message }); } }); // ========================================== // FRONTEND INTEGRATO (HTML / CSS / JS) // ========================================== app.get('/', (req, res) => { res.send(` Creator Hub & Stripe Connect Unificato

🔥 Profilo Follower & Stripe Connect

SISTEMA ATTIVO

1. Profilo Follower & Onboarding Stripe

2. Crea Contenuto / Servizio

3. Watchlist 50 Creator

🛒 Storefront & Piu recenti

Acquista direttamente dai profili connessi con trattenuta automatica.

Caricamento prodotti...

`); }); const PORT = process.env.PORT || 4280; app.listen(PORT, () => { console.log(`🚀 Server unificato avviato con successo su http://localhost:${PORT}`); });
